-- ===== AUTOEXEC SAFETY =====
repeat task.wait() until game:IsLoaded()

local Players = game:GetService("Players")
local StarterGui = game:GetService("StarterGui")
local HttpService = game:GetService("HttpService")

repeat task.wait() until Players.LocalPlayer
local player = Players.LocalPlayer

repeat task.wait() until player:FindFirstChild("PlayerGui")
local PlayerGui = player.PlayerGui

-- prevent double inject
if PlayerGui:FindFirstChild("MubashirAutoKickUI") then
	warn("[Mubashir] AutoKick already running")
	return
end

-- ===== SETTINGS =====
local KEYWORD = "you stole"
local KICK_MESSAGE = "mubashirs instant kick"
local POS_FILE = "mubashir_autokick_ui_pos.json"

local enabled = true
local connections = {}

-- ===== LOGIC =====
local function hasKeyword(text)
	if not enabled then return false end
	if typeof(text) ~= "string" then return false end
	return string.find(string.lower(text), KEYWORD) ~= nil
end

local function kick()
	if not enabled then return end
	pcall(function()
		player:Kick(KICK_MESSAGE)
	end)
end

local function watchObject(obj)
	if not (obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")) then
		return
	end

	if hasKeyword(obj.Text) then
		kick()
		return
	end

	table.insert(connections,
		obj:GetPropertyChangedSignal("Text"):Connect(function()
			if hasKeyword(obj.Text) then
				kick()
			end
		end)
	)
end

local function scan(parent)
	for _, obj in ipairs(parent:GetDescendants()) do
		watchObject(obj)
	end
end

local function watchGui(gui)
	scan(gui)
	table.insert(connections, gui.DescendantAdded:Connect(watchObject))
end

for _, gui in ipairs(PlayerGui:GetChildren()) do
	watchGui(gui)
end

table.insert(connections, PlayerGui.ChildAdded:Connect(watchGui))

-- ===== UI (DELAYED FOR AUTOEXEC) =====
task.wait(1)

local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "MubashirAutoKickUI"
ScreenGui.ResetOnSpawn = false
ScreenGui.Parent = PlayerGui

local Main = Instance.new("Frame")
Main.Size = UDim2.fromOffset(260, 140)
Main.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
Main.BorderSizePixel = 0
Main.Active = true
Main.Draggable = true
Main.Parent = ScreenGui

Instance.new("UICorner", Main).CornerRadius = UDim.new(0, 10)

local Title = Instance.new("TextLabel")
Title.Size = UDim2.new(1, 0, 0, 36)
Title.BackgroundColor3 = Color3.fromRGB(120, 0, 200)
Title.Text = "Mubashir's Auto Kick"
Title.TextColor3 = Color3.new(1, 1, 1)
Title.Font = Enum.Font.GothamBold
Title.TextSize = 16
Title.BorderSizePixel = 0
Title.Parent = Main
Instance.new("UICorner", Title).CornerRadius = UDim.new(0, 10)

local Toggle = Instance.new("TextButton")
Toggle.Size = UDim2.fromOffset(200, 44)
Toggle.Position = UDim2.fromOffset(30, 70)
Toggle.BackgroundColor3 = Color3.fromRGB(140, 0, 255)
Toggle.Text = "Auto Kick: ON"
Toggle.TextColor3 = Color3.new(1, 1, 1)
Toggle.Font = Enum.Font.GothamBold
Toggle.TextSize = 18
Toggle.BorderSizePixel = 0
Toggle.Parent = Main
Instance.new("UICorner", Toggle).CornerRadius = UDim.new(0, 8)

-- ===== LOAD POSITION =====
pcall(function()
	if isfile and isfile(POS_FILE) then
		local data = HttpService:JSONDecode(readfile(POS_FILE))
		Main.Position = UDim2.new(
			data.XScale, data.XOffset,
			data.YScale, data.YOffset
		)
	else
		Main.Position = UDim2.fromScale(0.03, 0.45)
	end
end)

-- ===== SAVE POSITION =====
local saveDebounce = false
Main:GetPropertyChangedSignal("Position"):Connect(function()
	if saveDebounce then return end
	saveDebounce = true
	task.delay(0.3, function()
		saveDebounce = false
		if writefile then
			local p = Main.Position
			writefile(POS_FILE, HttpService:JSONEncode({
				XScale = p.X.Scale,
				XOffset = p.X.Offset,
				YScale = p.Y.Scale,
				YOffset = p.Y.Offset
			}))
		end
	end)
end)

-- ===== TOGGLE =====
Toggle.MouseButton1Click:Connect(function()
	enabled = not enabled
	Toggle.Text = enabled and "Auto Kick: ON" or "Auto Kick: OFF"
	Toggle.BackgroundColor3 = enabled
		and Color3.fromRGB(140, 0, 255)
		or Color3.fromRGB(60, 60, 60)

	pcall(function()
		StarterGui:SetCore("SendNotification", {
			Title = "Mubashir's Auto Kick",
			Text = enabled and "ENABLED" or "DISABLED",
			Duration = 3
		})
	end)
end)

-- ===== EXECUTED MESSAGE =====
pcall(function()
	StarterGui:SetCore("SendNotification", {
		Title = "Mubashir's Auto Kick",
		Text = "Autoexec loaded & position restored.",
		Duration = 4
	})
end)

print("[Mubashir] AutoKick loaded successfully")
