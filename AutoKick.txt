local Players = game:GetService("Players")
local StarterGui = game:GetService("StarterGui")

local player = Players.LocalPlayer
local PlayerGui = player:WaitForChild("PlayerGui")

local KEYWORD = "you stole"
local KICK_MESSAGE = "Discord@rznnq"

local enabled = true
local connections = {}

-- 🔍 keyword check
local function hasKeyword(text)
	if not enabled then return false end
	if typeof(text) ~= "string" then return false end
	return string.find(string.lower(text), KEYWORD) ~= nil
end

-- 👢 kick
local function kick()
	if not enabled then return end
	pcall(function()
		player:Kick(KICK_MESSAGE)
	end)
end

-- 👀 watch text objects
local function watchObject(obj)
	if not (obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")) then
		return
	end

	if hasKeyword(obj.Text) then
		kick()
		return
	end

	local conn = obj:GetPropertyChangedSignal("Text"):Connect(function()
		if hasKeyword(obj.Text) then
			kick()
		end
	end)

	table.insert(connections, conn)
end

-- 🔎 scan gui
local function scan(parent)
	for _, obj in ipairs(parent:GetDescendants()) do
		watchObject(obj)
	end
end

-- 📺 watch new guis
local function watchGui(gui)
	scan(gui)
	local conn = gui.DescendantAdded:Connect(function(desc)
		watchObject(desc)
	end)
	table.insert(connections, conn)
end

-- 🧠 init existing guis
for _, gui in ipairs(PlayerGui:GetChildren()) do
	 watchGui(gui)
end

table.insert(connections,
	PlayerGui.ChildAdded:Connect(function(gui)
		watchGui(gui)
	end)
)

-- 🟣 UI BOX
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "MubashirAutoKickUI"
ScreenGui.ResetOnSpawn = false
ScreenGui.Parent = PlayerGui

local MainFrame = Instance.new("Frame")
MainFrame.Size = UDim2.fromOffset(260, 140)
MainFrame.Position = UDim2.fromScale(0.03, 0.45)
MainFrame.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
MainFrame.BorderSizePixel = 0
MainFrame.Parent = ScreenGui
MainFrame.Active = true
MainFrame.Draggable = true

local Corner = Instance.new("UICorner")
Corner.CornerRadius = UDim.new(0, 10)
Corner.Parent = MainFrame

-- 🟪 Title bar
local Title = Instance.new("TextLabel")
Title.Size = UDim2.new(1, 0, 0, 36)
Title.BackgroundColor3 = Color3.fromRGB(120, 0, 200)
Title.Text = "Mubashir's Auto Kick"
Title.TextColor3 = Color3.fromRGB(255, 255, 255)
Title.Font = Enum.Font.GothamBold
Title.TextSize = 16
Title.BorderSizePixel = 0
Title.Parent = MainFrame

local TitleCorner = Instance.new("UICorner")
TitleCorner.CornerRadius = UDim.new(0, 10)
TitleCorner.Parent = Title

-- 🔘 Toggle button
local Toggle = Instance.new("TextButton")
Toggle.Size = UDim2.fromOffset(200, 44)
Toggle.Position = UDim2.fromOffset(30, 70)
Toggle.BackgroundColor3 = Color3.fromRGB(140, 0, 255)
Toggle.TextColor3 = Color3.fromRGB(255, 255, 255)
Toggle.Font = Enum.Font.GothamBold
Toggle.TextSize = 18
Toggle.Text = "Auto Kick: ON"
Toggle.BorderSizePixel = 0
Toggle.Parent = MainFrame

local ToggleCorner = Instance.new("UICorner")
ToggleCorner.CornerRadius = UDim.new(0, 8)
ToggleCorner.Parent = Toggle

-- 🔘 click toggle
Toggle.MouseButton1Click:Connect(function()
	enabled = not enabled
	Toggle.Text = enabled and "Auto Kick: ON" or "Auto Kick: OFF"
	Toggle.BackgroundColor3 = enabled
		and Color3.fromRGB(140, 0, 255)
		or Color3.fromRGB(60, 60, 60)

	pcall(function()
		StarterGui:SetCore("SendNotification", {
			Title = "Mubashir's Auto Kick",
			Text = enabled and "Auto Kick ENABLED" or "Auto Kick DISABLED",
			Duration = 3
		})
	end)

	print("[Mubashir] AutoKick:", enabled and "ON" or "OFF")
end)

-- ✅ EXECUTED MESSAGE
pcall(function()
	StarterGui:SetCore("SendNotification", {
		Title = "Mubashir's Auto Kick",
		Text = "Script executed successfully.",
		Duration = 4
	})
end)

print("[Mubashir] AutoKick script executed.")
